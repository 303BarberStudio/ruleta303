// TEXTO RESPONSIVE (versión corregida) for (let i = 0; i < premios.length; i++) { const ang = i * anglePer + anglePer/2 + rot; // posición más centrada (evita que el texto se salga) const tx = center.x + Math.cos(ang) * (radius * 0.48); const ty = center.y + Math.sin(ang) * (radius * 0.48); const dynamicFont = Math.max(10, radius * 0.05); ctx.font = `bold ${dynamicFont}px Arial`; ctx.fillStyle = "#000"; ctx.textAlign = "center"; ctx.textBaseline = "middle"; const words = premios[i].split(" "); let line1 = ""; let line2 = ""; // forzar ruptura antes para títulos largos for (let w of words) { if (ctx.measureText(line1 + " " + w).width < radius * 0.45) { line1 += (line1 === "" ? "" : " ") + w; } else { line2 += (line2 === "" ? "" : " ") + w; } } ctx.save(); ctx.translate(tx, ty); let textAngle = ang + Math.PI/2; const deg = textAngle % (2*Math.PI); if (deg > Math.PI/2 && deg < 3*Math.PI/2) textAngle += Math.PI; ctx.rotate(textAngle); if (line2 === "") { ctx.fillText(line1, 0, 0); } else { ctx.fillText(line1, 0, -dynamicFont * 0.6); ctx.fillText(line2, 0, dynamicFont * 0.6); } ctx.restore(); }
